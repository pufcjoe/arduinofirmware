##############################################################
# Leonardo spoofed as Fnatic Bolt (1:1 mouse interface match)
#
# Real device descriptors (dumped from hardware):
#   bcdUSB=0x0200  bDeviceClass=0 (composite)
#   VID=0x2F0E  PID=0x0203  bcdDevice=0x0100
#   iManufacturer=1 ("Fnatic Gear")
#   iProduct=2 ("FNATIC BOLT Mouse")
#   iSerialNumber=3 ("000000000000")
#   bmAttributes=0xE0 (self-powered, remote wakeup)
#   MaxPower=100mA
#   Interface 0: Class=3 SubClass=1(Boot) Protocol=2(Mouse)
#     HID bcdHID=0x0111  ReportDescLen=73
#     Endpoint IN 0x81  maxPkt=64  interval=1
#   Interface 1: Class=3 SubClass=1 Protocol=0 (vendor)
#   Interface 2: Class=3 SubClass=1 Protocol=1 (keyboard)
#
# This board definition matches the mouse interface (IF0).
# CDC is disabled so the PC sees a pure HID device.
# To re-flash: double-tap the Leonardo reset button.
##############################################################

leonardo_bolt.name=Arduino Leonardo (Fnatic Bolt)

# Bootloader IDs stay as Leonardo so uploads still work
leonardo_bolt.vid.0=0x2341
leonardo_bolt.pid.0=0x0036
leonardo_bolt.vid.1=0x2341
leonardo_bolt.pid.1=0x8036
leonardo_bolt.upload_port.0.vid=0x2341
leonardo_bolt.upload_port.0.pid=0x0036
leonardo_bolt.upload_port.1.vid=0x2341
leonardo_bolt.upload_port.1.pid=0x8036
leonardo_bolt.upload_port.2.board=leonardo_bolt

leonardo_bolt.upload.tool=avrdude
leonardo_bolt.upload.tool.default=avrdude
leonardo_bolt.upload.tool.network=arduino_ota
leonardo_bolt.upload.protocol=avr109
leonardo_bolt.upload.maximum_size=28672
leonardo_bolt.upload.maximum_data_size=2560
leonardo_bolt.upload.speed=57600
leonardo_bolt.upload.disable_flushing=true
leonardo_bolt.upload.use_1200bps_touch=true
leonardo_bolt.upload.wait_for_upload_port=true

leonardo_bolt.bootloader.tool=avrdude
leonardo_bolt.bootloader.tool.default=avrdude
leonardo_bolt.bootloader.low_fuses=0xff
leonardo_bolt.bootloader.high_fuses=0xd8
leonardo_bolt.bootloader.extended_fuses=0xcb
leonardo_bolt.bootloader.file=caterina/Caterina-Leonardo.hex
leonardo_bolt.bootloader.unlock_bits=0x3F
leonardo_bolt.bootloader.lock_bits=0x2F

leonardo_bolt.build.mcu=atmega32u4
leonardo_bolt.build.f_cpu=16000000L
leonardo_bolt.build.vid=0x2F0E
leonardo_bolt.build.pid=0x0203
leonardo_bolt.build.usb_product="FNATIC BOLT Mouse"
leonardo_bolt.build.usb_manufacturer="Fnatic Gear"
leonardo_bolt.build.board=AVR_LEONARDO
leonardo_bolt.build.core=arduino
leonardo_bolt.build.variant=leonardo
leonardo_bolt.build.extra_flags={build.usb_flags} -DCDC_DISABLED -DUSB_CONFIG_POWER=100 -DUSB_CONFIG_ATTRS=0xE0 -DHID_INTERFACE_SUBCLASS=1 -DHID_INTERFACE_PROTOCOL=2 -DHID_ENDPOINT_INTERVAL=0x01 -DHID_ENDPOINT_SIZE=64 -DHID_BCD_VERSION_L=0x11 -DHID_BCD_VERSION_H=0x01
